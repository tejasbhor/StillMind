# =====================================================
# StillMind — Cloud Sync & Redeploy Script
# =====================================================

# 1. FIX OWNERSHIP (Required for OCI)
sudo chown -R $USER:$USER /opt/stillmind
git config --global --add safe.directory /opt/stillmind

cd /opt/stillmind

# 2. Pull latest code
echo "📥 Pulling latest code..."
git pull origin main

# 3. Stop StillMind containers
sudo docker compose down

# 4. Rebuild sequentially to conserve RAM
echo "🏗️ [1/2] Rebuilding Backend..."
sudo docker compose build backend

echo "🏗️ [2/2] Rebuilding Frontend..."
sudo docker compose build frontend

# 5. Start StillMind services
echo "🚀 Starting services..."
sudo docker compose up -d --remove-orphans

# 6. Database Migrations & Seeding (Critical!)
echo "🗃️ Running database migrations..."
# Note: We use 'alembic upgrade head'
sudo docker compose exec -T api uv run alembic upgrade head

echo "🌱 Seeding initial data..."
sudo docker compose exec -T api uv run python scripts/seed_all.py

echo "✅ ALL DONE! StillMind is live at https://stillmind.civiclens.space"
